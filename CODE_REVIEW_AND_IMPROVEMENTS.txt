from statsmodels.tsa.stattools import acf

def find_optimal_lags(series, max_lag=96, threshold=0.3):
    """基于ACF选择重要的滞后期"""
    acf_values = acf(series, nlags=max_lag)
    significant_lags = np.where(np.abs(acf_values) > threshold)[0]
    return tuple(significant_lags)

def adaptive_rolling_mean(series, hour):
    """根据时段选择窗口大小"""
    if 0 <= hour < 6:  # 夜间：稳定，用大窗口
        window = 32
    elif 6 <= hour < 9:  # 早高峰：变化快，用小窗口
        window = 8
    elif 9 <= hour < 17:  # 白天：中等窗口
        window = 16
    else:  # 晚高峰
        window = 12
    return series.rolling(window).mean()

def smart_fill_missing(df, column, device_type):
    """智能填充策略"""

    if device_type == 'blast_furnace':
        # 高炉：连续运行，用线性插值
        df[column] = df[column].interpolate(method='linear', limit=8)

    elif device_type == 'converter':
        # 转炉：间歇运行，用前向填充
        df[column] = df[column].fillna(method='ffill', limit=4)

    elif device_type == 'air_heater':
        # 热风炉：周期运行，用季节性
        # 相同hour的历史中位数
        hour_median = df.groupby(df['datetime'].dt.hour)[column].transform('median')
        df[column] = df[column].fillna(hour_median)

    # 最后用全局中位数兜底
    df[column] = df[column].fillna(df[column].median())

    return df

### 问题 2.2: 没有异常值检测
```python
# 当前代码没有明确的异常值检测和修正
```

❌ 问题：
- 传感器故障导致的异常值会影响模型
- 数据录入错误没有被捕获
- 物理不可能的值（负数、超容量）存在

✅ 建议：
```python
def detect_and_fix_outliers(df):
    """异常值检测与修正"""

    # 1. 物理约束检查
    df['generator_1'] = df['generator_1'].clip(0, 120)  # MW
    df['generator_all'] = df['generator_all'].clip(0, 200)
    df['blast_furnace_gas_holder_2'] = df['blast_furnace_gas_holder_2'].clip(0, 100000)  # m³

    # 2. 变化率检查（15分钟不应变化超过30%）
    for col in ['generator_1', 'generator_all']:
        change_rate = df[col].pct_change().abs()
        outliers = change_rate > 0.3

        # 标记异常值
        df[f'{col}_outlier_flag'] = outliers.astype(int)

        # 用插值修正
        df.loc[outliers, col] = np.nan
        df[col] = df[col].interpolate(method='linear', limit=4)

    # 3. 统计异常检测（滑动窗口 + IQR）
    for col in ['blast_furnace_gas_holder_2', 'generator_use_blast_furnace_gas']:
        rolling = df[col].rolling(96, min_periods=24)  # 1天窗口
        Q1 = rolling.quantile(0.25)
        Q3 = rolling.quantile(0.75)
        IQR = Q3 - Q1
        lower = Q1 - 3 * IQR
        upper = Q3 + 3 * IQR

        outliers = (df[col] < lower) | (df[col] > upper)
        df[f'{col}_outlier_flag'] = outliers.astype(int)

        # Winsorize（限制在合理范围）
        df[col] = df[col].clip(lower, upper)

    return df
```

预期改进：0.10-0.30pp（重要！）

### A. 物理约束后处理
```python
def apply_physical_constraints(predictions, current_state):
    """确保预测符合物理规律"""

    pred_fixed = predictions.copy()

    # 1. 容量约束
    pred_fixed['generator_1'] = pred_fixed['generator_1'].clip(0, 120)
    pred_fixed['generator_all'] = pred_fixed['generator_all'].clip(0, 200)

    # 2. 一致性约束
    # generator_all >= generator_1
    pred_fixed['generator_all'] = np.maximum(
        pred_fixed['generator_all'],
        pred_fixed['generator_1']
    )

    # 3. 平滑约束（连续时间步变化不能太大）
    for h in range(2, 9):
        max_change = 20  # MW per 15min
        prev = pred_fixed[f'h{h-1}']
        curr = pred_fixed[f'h{h}']

        # 限制变化幅度
        curr_fixed = np.clip(curr, prev - max_change, prev + max_change)
        pred_fixed[f'h{h}'] = curr_fixed

    # 4. 能量守恒约束
    # 预测的发电量不能超过可用气量
    available_gas = current_state['holder'] + current_state['bf_supply'] * 2  # 2小时
    max_power_from_gas = available_gas * 0.5  # 转换效率

    pred_fixed['generator_all'] = np.minimum(
        pred_fixed['generator_all'],
        max_power_from_gas
    )

    return pred_fixed
```

### B. 时间序列平滑
```python
from scipy.signal import savgol_filter

def smooth_predictions(predictions):
    """平滑预测序列"""

    # 对8个时间步应用Savitzky-Golay滤波
    # 保留趋势，去除噪声
    smoothed = {}
    for target in ['generator_1', 'generator_all']:
        raw_seq = [predictions[f'{target}_h{h}'] for h in range(1, 9)]
        smooth_seq = savgol_filter(raw_seq, window_length=5, polyorder=2)

        for h in range(1, 9):
            smoothed[f'{target}_h{h}'] = smooth_seq[h-1]

    return smoothed
```

### C. 预测校准
```python
def calibrate_predictions(predictions, validation_errors):
    """基于验证集误差校准预测"""

    # 计算每个时间步的系统偏差
    bias_h = {}
    for h in range(1, 9):
        bias_h[h] = validation_errors[f'h{h}'].mean()

    # 校准
    calibrated = predictions.copy()
    for h in range(1, 9):
        calibrated[f'h{h}'] = calibrated[f'h{h}'] - bias_h[h]

    return calibrated
```
