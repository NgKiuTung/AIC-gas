```python
def preprocess_with_outlier_detection(df):
    \"\"\"Step 1: 异常值检测与修正\"\"\"

    # 物理约束
    df['generator_1'] = df['generator_1'].clip(0, 120)
    df['holder'] = df['holder'].clip(0, 100000)

    # 变化率检查
    max_rate = 0.3
    for col in ['generator_1', 'generator_all']:
        change = df[col].pct_change().abs()
        outliers = change > max_rate

        # 用插值修正
        df.loc[outliers, col] = np.nan
        df[col] = df[col].interpolate(method='linear', limit=4)

    return df

def add_physical_features(df):
    \"\"\"Step 2: 添加物理约束特征\"\"\"

    # 能量守恒
    df['energy_balance'] = (
        df['bf_supply'] - df['total_use'] - df['holder_delta']
    )

    # 设备负载率
    df['gen1_load'] = df['generator_1'] / 120
    df['holder_fill'] = df['holder'] / 100000

    # 效率指标
    df['power_per_gas'] = df['generator_all'] / (df['total_use'] + 1)

    return df

def enhanced_preprocessing_pipeline(tables, price_lookup):
    \"\"\"完整优化流程\"\"\"

    # 1. 基础处理
    merged = merge_raw_tables(tables)

    # 2. 异常值检测
    cleaned = preprocess_with_outlier_detection(merged)

    # 3. 缺失值填充（优化后）
    filled = advanced_fill_missing(cleaned)

    # 4. 物理特征
    enriched = add_physical_features(filled)

    # 5. 基础特征工程
    features = build_inference_feature_frame(enriched)

    # 6. 未来价格特征
    with_prices = add_future_price_features(features, price_lookup)

    return with_prices
```
